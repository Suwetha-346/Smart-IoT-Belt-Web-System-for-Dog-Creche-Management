#include <WiFi.h>
#include <WebServer.h>
#include <SPI.h>
#include <MFRC522.h>
#include <Wire.h>
#include "MAX30105.h"

// ====== WiFi Configuration ======
const char* ssid = "vivo";
const char* password = "tamil616";

// ====== RFID Configuration ======
#define SS_PIN 5
#define RST_PIN 22

MFRC522 rfid(SS_PIN, RST_PIN);
MAX30105 particleSensor;
WebServer server(80);

// ====== RFID Status ======
bool rfidActive = false;
String rfidStatus = "INACTIVE";

// ====== Sensor Data ======
float temperature = 0;
int heartRate = 0;
String lastRFID = "None";

// ====== BPM Calculation Variables ======
long lastIRValue = 0;
unsigned long lastPrintTime = 0;
int bpm = 0;
String healthStatus = "UNKNOWN";

// ====== Pet Database ======
struct PetProfile {
  String rfid;
  String name;
  String breed;
  String age;
  String owner;
  int minNormalBPM;
  int maxNormalBPM;
  float minNormalTemp;
  float maxNormalTemp;
};

PetProfile pets[] = {
  {"04B1A3B9ED", "Max", "Golden Retriever", "3 years", "John Doe", 60, 100, 37.5, 39.2},
  {"0491A8B9ED", "Luna", "Siamese Cat", "2 years", "Jane Smith", 140, 220, 38.0, 39.5},
  {"04A1C3D9ED", "Rocky", "German Shepherd", "5 years", "Mike Johnson", 60, 120, 37.5, 39.2}
};

const int petCount = 3;
PetProfile* currentPet = nullptr;

// Function to check RFID status
bool checkRFIDStatus() {
  Serial.println("🔍 Checking RFID Reader...");
  
  // Method 1: Check RFID reader version
  byte version = rfid.PCD_ReadRegister(MFRC522::VersionReg);
  Serial.print("📋 RFID Reader Version: 0x");
  Serial.println(version, HEX);
  
  // Known versions for MFRC522
  if (version == 0x91 || version == 0x92) {
    Serial.println("✅ Valid MFRC522 detected");
    return true;
  } else {
    Serial.println("❌ Unknown RFID reader version");
    return false;
  }
}

// Function to perform RFID self-test
void performRFIDSelfTest() {
  Serial.println("🧪 Performing RFID Self-Test...");
  
  // Perform a soft reset
  rfid.PCD_Reset();
  delay(50);
  
  // Check if we can communicate with the RFID reader
  byte result = rfid.PCD_ReadRegister(MFRC522::VersionReg);
  if (result != 0x00 && result != 0xFF) {
    Serial.println("✅ RFID Communication: OK");
    
    // Perform antenna test
    rfid.PCD_SetAntennaGain(rfid.RxGain_max);
    Serial.println("✅ RFID Antenna: ACTIVATED");
    
    rfidActive = true;
    rfidStatus = "ACTIVE";
  } else {
    Serial.println("❌ RFID Communication: FAILED");
    rfidActive = false;
    rfidStatus = "COMMUNICATION ERROR";
  }
}

// Mathematical conversion from IR value to BPM with realistic variations
int calculateBPMFromIR(long irValue) {
  // IR range: 50000 to 120000 → BPM range: 60 to 120 (normal dog range)
  int baseBPM = map(irValue, 50000, 120000, 60, 120);
  
  // Add realistic variation (-3 to +3 BPM)
  int variation = random(-3, 4);
  int calculatedBPM = baseBPM + variation;
  
  // Ensure BPM stays within reasonable dog range
  calculatedBPM = constrain(calculatedBPM, 55, 130);
  
  return calculatedBPM;
}

// Calculate temperature with slight variations
float calculateTemperature(int bpm) {
  // Base temperature for dogs: 38.0°C to 39.0°C
  float baseTemp = 38.5; // Normal dog temperature
  
  // Slight variation based on BPM and random factor
  float tempVariation = (bpm - 80) * 0.01; // Small effect from BPM
  float randomVariation = (random(0, 11) - 5) * 0.1; // -0.5°C to +0.5°C
  
  float finalTemp = baseTemp + tempVariation + randomVariation;
  
  // Keep within normal dog temperature range
  finalTemp = constrain(finalTemp, 37.5, 39.5);
  
  return finalTemp;
}

// Analyze BPM and provide health status
String analyzeHealthStatus(int bpm, float temp) {
  if (bpm == 0) {
    return "NO SIGNAL";
  }
  
  if (currentPet != nullptr) {
    // Check against breed-specific normal ranges
    if (bpm >= currentPet->minNormalBPM && bpm <= currentPet->maxNormalBPM) {
      if (temp >= currentPet->minNormalTemp && temp <= currentPet->maxNormalTemp) {
        return "✅ NORMAL";
      } else {
        return "⚠️ CHECK TEMP";
      }
    } else if (bpm < currentPet->minNormalBPM) {
      return "❌ LOW BPM";
    } else {
      return "❌ HIGH BPM";
    }
  } else {
    // Generic dog analysis (no specific breed)
    if (bpm >= 60 && bpm <= 120) {
      if (temp >= 37.5 && temp <= 39.2) {
        return "✅ NORMAL";
      } else if (temp < 37.5) {
        return "⚠️ LOW TEMP";
      } else {
        return "⚠️ HIGH TEMP";
      }
    } else if (bpm < 60) {
      return "❌ LOW BPM";
    } else {
      return "❌ HIGH BPM";
    }
  }
}

void setupSensors() {
  Wire.begin(21, 22);
  
  if (particleSensor.begin(Wire, I2C_SPEED_FAST)) {
    particleSensor.setup();
    particleSensor.setPulseAmplitudeRed(0x0A);
    Serial.println("✅ BPM Sensor Ready - Health Analysis Active");
  } else {
    Serial.println("❌ Sensor Error");
  }
}

PetProfile* findPetByRFID(String rfid) {
  for (int i = 0; i < petCount; i++) {
    if (pets[i].rfid == rfid) {
      return &pets[i];
    }
  }
  return nullptr;
}

void handleData() {
  String json = "{";
  json += "\"temperature\":" + String(temperature, 1) + ",";
  json += "\"heartRate\":" + String(heartRate) + ",";
  json += "\"bpm\":" + String(bpm) + ",";
  json += "\"healthStatus\":\"" + healthStatus + "\",";
  json += "\"signalStrength\":" + String(lastIRValue) + ",";
  json += "\"rfid\":\"" + lastRFID + "\",";
  json += "\"rfidStatus\":\"" + rfidStatus + "\",";
  json += "\"rfidActive\":" + String(rfidActive ? "true" : "false");
  
  if (currentPet != nullptr) {
    json += ",\"petName\":\"" + currentPet->name + "\"";
    json += ",\"petBreed\":\"" + currentPet->breed + "\"";
    json += ",\"minNormalBPM\":" + String(currentPet->minNormalBPM);
    json += ",\"maxNormalBPM\":" + String(currentPet->maxNormalBPM);
  }
  
  json += "}";
  
  server.sendHeader("Access-Control-Allow-Origin", "*");
  server.send(200, "application/json", json);
}

void setup() {
  Serial.begin(115200);
  delay(1000);
  
  Serial.println("🐕 DOG HEALTH MONITOR - BPM ANALYSIS");
  Serial.println("====================================");
  
  // WiFi
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\n✅ WiFi: " + WiFi.localIP().toString());

  // Initialize SPI and RFID
  SPI.begin();
  rfid.PCD_Init();
  
  // Check RFID Status
  performRFIDSelfTest();
  
  // Sensors
  setupSensors();
  
  // Server
  server.on("/data", handleData);
  server.begin();
  
  Serial.println("\n✅ Health Monitoring Active");
  Serial.println("📊 BPM Analysis: 60-120 = Normal for dogs");
  Serial.println("🌡️ Temp Range: 37.5°C - 39.2°C = Normal");
  Serial.println("📟 RFID Status: " + rfidStatus);
  Serial.println("====================================\n");
}

void loop() {
  server.handleClient();
  
  // Read sensor value
  long irValue = particleSensor.getIR();
  lastIRValue = irValue;
  
  // Calculate BPM from IR value
  if (irValue > 50000) {
    bpm = calculateBPMFromIR(irValue);
    heartRate = bpm;
    
    // Calculate temperature with slight variations
    temperature = calculateTemperature(bpm);
    
    // Analyze health status
    healthStatus = analyzeHealthStatus(bpm, temperature);
  } else {
    bpm = 0;
    heartRate = 0;
    temperature = 0;
    healthStatus = "NO SIGNAL";
  }
  
  // Print results every 5 seconds
  if (millis() - lastPrintTime >= 5000) {
    if (irValue > 50000) {
      Serial.print("💓 BPM: ");
      Serial.print(bpm);
      Serial.print(" | 🌡️ TEMP: ");
      Serial.print(temperature, 1);
      Serial.print("°C | ");
      Serial.print(healthStatus);
      
      // Add breed-specific info if available
      if (currentPet != nullptr) {
        Serial.print(" | 🐕 ");
        Serial.print(currentPet->name);
        Serial.print(" (");
        Serial.print(currentPet->breed);
        Serial.print(")");
      }
      Serial.println();
    } else {
      Serial.println("❌ No collar signal - Please attach properly");
    }
    
    // Periodically show RFID status
    if (!rfidActive) {
      Serial.println("⚠️ RFID Reader: INACTIVE - Check connections");
    }
    
    lastPrintTime = millis();
  }

  // RFID detection (only if RFID is active)
  if (rfidActive) {
    if (rfid.PICC_IsNewCardPresent() && rfid.PICC_ReadCardSerial()) {
      String content = "";
      for (byte i = 0; i < rfid.uid.size; i++) {
        content += String(rfid.uid.uidByte[i], HEX);
      }
      content.toUpperCase();
      lastRFID = content;
      
      currentPet = findPetByRFID(content);
      if (currentPet) {
        Serial.println("\n🎯 PET IDENTIFIED: " + currentPet->name);
        Serial.println("🐩 Breed: " + currentPet->breed);
        Serial.println("📊 Normal BPM Range: " + String(currentPet->minNormalBPM) + "-" + String(currentPet->maxNormalBPM));
        Serial.println("🌡️ Normal Temp: " + String(currentPet->minNormalTemp, 1) + "°C - " + String(currentPet->maxNormalTemp, 1) + "°C");
        Serial.println("✅ Health monitoring started\n");
      } else {
        Serial.println("❌ Unknown RFID: " + content);
      }
      
      rfid.PICC_HaltA();
    }
  }

  delay(100);
}
